@dataclass
class VideoMetadata:
    """Metadata extracted from video filename"""
    filepath: str
    filename: str
    corpus: str
    speaker: str
    clip_id: str
    category: str
    subtype: str
    is_mirror: bool
    training_label: str = ""
    speaker_id: str = ""
    
    def __post_init__(self):
        self.speaker_id = f"{self.corpus}_{self.speaker}"
        self.training_label = self._get_training_label()
    
    def _get_training_label(self) -> str:
        """Determine training label based on category and subtype"""
        if self.category == 'nogesture':
            return 'NoGesture'
        elif self.subtype.lower() in ['move', 'adaptor']:
            return 'Move'
        elif self.subtype.lower() == 'objman':
            return 'Objman'
        else:
            return 'Gesture'
    
    def to_dict(self) -> Dict[str, Any]:
        return {
            'filepath': self.filepath,
            'filename': self.filename,
            'corpus': self.corpus,
            'speaker': self.speaker,
            'clip_id': self.clip_id,
            'category': self.category,
            'subtype': self.subtype,
            'is_mirror': self.is_mirror,
            'training_label': self.training_label,
            'speaker_id': self.speaker_id
        }


def parse_filename(filepath: str) -> Optional[VideoMetadata]:
    """Parse video filename to extract metadata."""
    filename = os.path.basename(filepath)
    
    is_mirror = "_mirror" in filename
    clean_name = filename.replace("_mirror", "").replace(".mp4", "")
    
    parts = clean_name.split("_")
    
    if len(parts) < 4:
        print(f"Warning: Cannot parse filename (too few parts): {filename}")
        return None
    
    corpus = parts[0]
    
    # Find category index
    category_idx = None
    for i, p in enumerate(parts):
        if p.lower() in ['gesture', 'nogesture']:
            category_idx = i
            break
    
    if category_idx is None or category_idx < 2:
        print(f"Warning: Cannot find category in: {filename}")
        return None
    
    clip_id = parts[category_idx - 1]
    
    if category_idx > 2:
        speaker = "_".join(parts[1:category_idx-1])
    else:
        speaker = parts[1]
    
    category = parts[category_idx].lower()
    subtype = parts[category_idx + 1] if category_idx + 1 < len(parts) else "NA"
    
    return VideoMetadata(
        filepath=filepath,
        filename=filename,
        corpus=corpus,
        speaker=speaker,
        clip_id=clip_id,
        category=category,
        subtype=subtype,
        is_mirror=is_mirror
    )

def calculate_angle(p1: Tuple[float, float], p2: Tuple[float, float], p3: Tuple[float, float]) -> float:
    """Calculate angle between three points in degrees."""
    v1 = np.array(p1) - np.array(p2)
    v2 = np.array(p3) - np.array(p2)
    cos_angle = np.dot(v1, v2) / (np.linalg.norm(v1) * np.linalg.norm(v2) + 1e-6)
    cos_angle = np.clip(cos_angle, -1.0, 1.0)
    return np.degrees(np.arccos(cos_angle))


def extract_hand_features(hand_landmarks, wrist_pos: Tuple[float, float], 
                         norm_dist: float, w: int, h: int) -> List[float]:
    """Extract 10 hand features from MediaPipe hand landmarks."""
    if hand_landmarks is None or norm_dist == 0:
        return [0.0] * 10
    
    lm = hand_landmarks.landmark
    
    wrist = (lm[0].x * w, lm[0].y * h)
    thumb_tip = (lm[4].x * w, lm[4].y * h)
    index_tip = (lm[8].x * w, lm[8].y * h)
    middle_tip = (lm[12].x * w, lm[12].y * h)
    ring_tip = (lm[16].x * w, lm[16].y * h)
    pinky_tip = (lm[20].x * w, lm[20].y * h)
    
    index_mcp = (lm[5].x * w, lm[5].y * h)
    middle_mcp = (lm[9].x * w, lm[9].y * h)
    ring_mcp = (lm[13].x * w, lm[13].y * h)
    pinky_mcp = (lm[17].x * w, lm[17].y * h)
    
    dists = [
        np.linalg.norm(np.array(wrist) - np.array(tip)) / norm_dist
        for tip in [thumb_tip, index_tip, middle_tip, ring_tip, pinky_tip]
    ]
    
    angles = [
        calculate_angle(thumb_tip, wrist, index_tip) / 180.0,
        calculate_angle(index_tip, index_mcp, middle_tip) / 180.0,
        calculate_angle(middle_tip, middle_mcp, ring_tip) / 180.0,
        calculate_angle(ring_tip, ring_mcp, pinky_tip) / 180.0,
    ]
    
    extensions = []
    for tip, mcp in [(index_tip, index_mcp), (middle_tip, middle_mcp), 
                     (ring_tip, ring_mcp), (pinky_tip, pinky_mcp)]:
        tip_dist = np.linalg.norm(np.array(wrist) - np.array(tip))
        mcp_dist = np.linalg.norm(np.array(wrist) - np.array(mcp))
        extensions.append(tip_dist / (mcp_dist + 1e-6))
    openness = np.mean(extensions)
    
    return dists + angles + [openness]


def extract_visibility_features(results) -> List[float]:
    """Extract 6 visibility/confidence features."""
    visibility = [0.0] * 6
    
    if results.pose_landmarks:
        pose_lm = results.pose_landmarks.landmark
        visibility[0] = pose_lm[15].visibility if len(pose_lm) > 15 else 0.0
        visibility[1] = pose_lm[16].visibility if len(pose_lm) > 16 else 0.0
        visibility[2] = pose_lm[13].visibility if len(pose_lm) > 13 else 0.0
        visibility[3] = pose_lm[14].visibility if len(pose_lm) > 14 else 0.0
    
    visibility[4] = 1.0 if results.left_hand_landmarks else 0.0
    visibility[5] = 1.0 if results.right_hand_landmarks else 0.0
    
    return visibility


def extract_move_distinguishing_features(body: Dict, nose_2d: Tuple[float, float], 
                                         norm_dist: float) -> List[float]:
    """Extract 6 move-distinguishing features."""
    features = [0.0] * 6
    
    lw, rw = body.get('LEFT_WRIST'), body.get('RIGHT_WRIST')
    ls, rs = body.get('LEFT_SHOULDER'), body.get('RIGHT_SHOULDER')
    li, ri = body.get('LEFT_INDEX'), body.get('RIGHT_INDEX')

    # 1. Hand-to-face distance
    l_dist = np.linalg.norm(np.array(li) - np.array(nose_2d)) / norm_dist if li else 99
    r_dist = np.linalg.norm(np.array(ri) - np.array(nose_2d)) / norm_dist if ri else 99
    min_dist = min(l_dist, r_dist)
    features[0] = min_dist if min_dist < 99 else 0.0

    # 2. Contact threshold
    features[1] = 1.0 if min_dist < 0.35 else 0.0 

    # 3. Min height relative to shoulders
    if lw and ls and rw and rs:
        features[2] = min((lw[1]-ls[1])/norm_dist, (rw[1]-rs[1])/norm_dist)

    # 4 & 5. Wrists above shoulders
    features[3] = 1.0 if (lw and ls and lw[1] < ls[1]) else 0.0
    features[4] = 1.0 if (rw and rs and rw[1] < rs[1]) else 0.0

    # 6. Symmetry
    if lw and rw and ls and rs:
        center_x = (ls[0] + rs[0]) / 2
        symm = abs(abs(lw[0]-center_x) - abs(rw[0]-center_x)) / norm_dist
        features[5] = max(0, 1.0 - symm)
    
    return features


def extract_world_landmarks_with_visibility(results) -> Optional[List[float]]:
    """Extract upper body world landmarks with visibility (23 × 4 = 92 features)."""
    if not results.pose_world_landmarks:
        return None
    
    features = []
    for idx in UPPER_BODY_INDICES:
        if idx < len(results.pose_world_landmarks.landmark):
            lm = results.pose_world_landmarks.landmark[idx]
            features.extend([lm.x, lm.y, lm.z, lm.visibility])
        else:
            features.extend([0.0, 0.0, 0.0, 0.0])
    
    return features


def extract_basic_features(results, w: int, h: int) -> Tuple[List[float], Dict, Tuple, float]:
    """Extract 29 basic features."""
    pose_landmarks = results.pose_landmarks.landmark
    nose_pose = pose_landmarks[0]
    nose_2d = (nose_pose.x * w, nose_pose.y * h)
    nose_3d = (nose_pose.x * w, nose_pose.y * h, nose_pose.z * 3000)

    face_2d, face_3d = [], []
    chin, top_head = (0, 0), (0, 0)
    left_inner_eye, left_brow = (0, 0), (0, 0)
    right_inner_eye, right_brow = (0, 0), (0, 0)
    left_mouth, right_mouth = (0, 0), (0, 0)
    upper_lip, lower_lip = (0, 0), (0, 0)

    for idx, lm in enumerate(results.face_landmarks.landmark):
        if idx in [33, 263, 1, 61, 291, 199]:
            x, y = int(lm.x * w), int(lm.y * h)
            face_2d.append([x, y])
            face_3d.append([x, y, lm.z])
        
        if idx == 152: chin = (lm.x * w, lm.y * h)
        elif idx == 10: top_head = (lm.x * w, lm.y * h)
        elif idx == 133: left_inner_eye = (lm.x * w, lm.y * h)
        elif idx == 443: left_brow = (lm.x * w, lm.y * h)
        elif idx == 223: right_brow = (lm.x * w, lm.y * h)
        elif idx == 362: right_inner_eye = (lm.x * w, lm.y * h)
        elif idx == 87: left_mouth = (lm.x * w, lm.y * h)
        elif idx == 308: right_mouth = (lm.x * w, lm.y * h)
        elif idx == 13: upper_lip = (lm.x * w, lm.y * h)
        elif idx == 14: lower_lip = (lm.x * w, lm.y * h)

    face_2d = np.array(face_2d, dtype=np.float64)
    face_3d = np.array(face_3d, dtype=np.float64)
    focal_length = w
    cam_matrix = np.array([[focal_length, 0, h / 2], [0, focal_length, w / 2], [0, 0, 1]])
    success, rot_vec, _ = cv2.solvePnP(face_3d, face_2d, cam_matrix, np.zeros((4, 1)))
    
    if success:
        rmat, _ = cv2.Rodrigues(rot_vec)
        angles, _, _, _, _, _ = cv2.RQDecomp3x3(rmat)
        xrot, yrot, zrot = angles[0] * w, angles[1] * h, angles[2] * 3000
    else:
        xrot, yrot, zrot = 0.0, 0.0, 0.0

    norm_dist = np.linalg.norm(np.array(chin) - np.array(top_head))
    if norm_dist == 0: norm_dist = 1
    
    body = {}
    for idx, lm in enumerate(pose_landmarks):
        if idx < len(BODY_LANDMARKS):
            body[BODY_LANDMARKS[idx]] = (lm.x * w, lm.y * h)

    def sd(p1, p2): 
        return np.linalg.norm(np.array(p1) - np.array(p2)) / norm_dist if (p1 and p2) else 0.0

    features = [
        xrot, yrot, zrot, nose_3d[0], nose_3d[1], nose_3d[2], norm_dist,
        sd(left_inner_eye, left_brow), sd(right_inner_eye, right_brow),
        sd(left_mouth, right_mouth), sd(upper_lip, lower_lip),
        sd(body.get('LEFT_WRIST'), body.get('RIGHT_WRIST')),
        sd(body.get('LEFT_ELBOW'), body.get('RIGHT_ELBOW')),
        sd(body.get('LEFT_ELBOW'), body.get('LEFT_SHOULDER')),
        sd(body.get('RIGHT_ELBOW'), body.get('RIGHT_SHOULDER')),
        sd(body.get('LEFT_WRIST'), body.get('LEFT_SHOULDER')),
        sd(body.get('RIGHT_WRIST'), body.get('RIGHT_SHOULDER')),
        sd(body.get('LEFT_SHOULDER'), body.get('LEFT_EAR')),
        sd(body.get('RIGHT_SHOULDER'), body.get('RIGHT_EAR')),
        sd(body.get('LEFT_THUMB'), body.get('LEFT_INDEX')),
        sd(body.get('RIGHT_THUMB'), body.get('RIGHT_INDEX')),
        sd(body.get('LEFT_THUMB'), body.get('LEFT_PINKY')),
        sd(body.get('RIGHT_THUMB'), body.get('RIGHT_PINKY')),
        (body.get('LEFT_WRIST')[0] - body.get('LEFT_ELBOW')[0])/norm_dist if (body.get('LEFT_WRIST') and body.get('LEFT_ELBOW')) else 0,
        (body.get('RIGHT_WRIST')[0] - body.get('RIGHT_ELBOW')[0])/norm_dist if (body.get('RIGHT_WRIST') and body.get('RIGHT_ELBOW')) else 0,
        (body.get('LEFT_WRIST')[1] - body.get('LEFT_ELBOW')[1])/norm_dist if (body.get('LEFT_WRIST') and body.get('LEFT_ELBOW')) else 0,
        (body.get('RIGHT_WRIST')[1] - body.get('RIGHT_ELBOW')[1])/norm_dist if (body.get('RIGHT_WRIST') and body.get('RIGHT_ELBOW')) else 0,
        sd(body.get('LEFT_INDEX'), nose_2d),
        sd(body.get('RIGHT_INDEX'), nose_2d),
    ]
    
    return features, body, nose_2d, norm_dist

def process_single_video(video_path: str, max_frames: int = 800, target_fps: int = 25,
                        min_detection_pct: float = 0.3) -> Optional[Dict[str, Any]]:
    """Process a single video and extract all three feature sets."""
    metadata = parse_filename(video_path)
    if metadata is None:
        return None
    
    cap = cv2.VideoCapture(video_path)
    if not cap.isOpened():
        print(f"Error opening: {video_path}")
        return None
    
    fps = cap.get(cv2.CAP_PROP_FPS)
    total_frames = int(cap.get(cv2.CAP_PROP_FRAME_COUNT))
    w = int(cap.get(cv2.CAP_PROP_FRAME_WIDTH))
    h = int(cap.get(cv2.CAP_PROP_FRAME_HEIGHT))
    frame_interval = max(1, int(fps / target_fps))
    
    basic_features = []
    extended_features = []
    world_features = []
    timestamps = []
    
    frame_count = 0
    attempted_frames = 0
    detected_frames = 0
    
    with mp_holistic.Holistic(
        min_detection_confidence=0.5,
        min_tracking_confidence=0.5,
        model_complexity=1
    ) as holistic:
        
        while cap.isOpened() and len(basic_features) < max_frames:
            ret, frame = cap.read()
            if not ret:
                break
            
            if frame_count % frame_interval != 0:
                frame_count += 1
                continue
            
            attempted_frames += 1
            frame_rgb = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)
            results = holistic.process(frame_rgb)
            
            if results.face_landmarks and results.pose_landmarks:

def world_to_screen(x: float, y: float, z: float, 
                   view: str, width: int, height: int,
                   scale: float = 400, 
                   offset_x: float = 0.5, offset_y: float = 0.5) -> Tuple[int, int]:
    """
    Transform world coordinates to screen coordinates for different views.
    
    World coordinate system (MediaPipe):
    - X: Left(-) to Right(+)
    - Y: Up(-) to Down(+)  [negative = above hip, positive = below hip]
    - Z: Towards camera(-) to Away(+)
    
    Note: Coordinates are pre-centered (centroid subtracted) so offset=0.5 centers.
    
    Args:
        x, y, z: World coordinates (centered, in meters)
        view: 'front', 'left', or 'right'
        width, height: Screen dimensions
        scale: Scaling factor (pixels per meter)
        offset_x, offset_y: Offset ratios (0.5 = center)
    
    Returns:
        (screen_x, screen_y) pixel coordinates
    """
    # Scale Z more aggressively for side views (Z is typically small)
    z_scale = scale * 1.0
    
    if view == 'front':
        # Front view: X horizontal, Y vertical
        screen_x = int(width * offset_x + x * scale)
        screen_y = int(height * offset_y + y * scale)
    
    elif view == 'left':
        # Left side view: Z horizontal, Y vertical
        screen_x = int(width * offset_x + z * z_scale)
        screen_y = int(height * offset_y + y * scale)
    
    elif view == 'right':
        # Right side view: Z horizontal (flipped), Y vertical
        screen_x = int(width * offset_x - z * z_scale)
        screen_y = int(height * offset_y + y * scale)
    
    else:
        raise ValueError(f"Unknown view: {view}")
    
    return screen_x, screen_y


def extract_landmark_coords(frame_features: np.ndarray) -> List[Tuple[float, float, float, float]]:
    """
    Extract (x, y, z, visibility) for each of 23 landmarks from 92-feature vector.
    
    Args:
        frame_features: Array of 92 features (23 landmarks × 4 values)
    
    Returns:
        List of 23 tuples: (x, y, z, visibility)
    """
    landmarks = []
    for i in range(23):
        idx = i * 4
        x = frame_features[idx]
        y = frame_features[idx + 1]
        z = frame_features[idx + 2]
        vis = frame_features[idx + 3]
        landmarks.append((x, y, z, vis))
    return landmarks

def render_skeleton_frame(landmarks: List[Tuple[float, float, float, float]], 
                         config: RenderConfig) -> np.ndarray:
    """
    Render a single frame with 3 views side by side.
    
    Args:
        landmarks: List of 23 (x, y, z, visibility) tuples
        config: Rendering configuration
    
    Returns:
        Combined frame (height × width*3 × 3)
    """
    w, h = config.frame_width, config.frame_height
    views = ['front', 'left', 'right']
    view_frames = []
    
    # Compute centroid for centering (use visible landmarks only)
    xs = [lm[0] for lm in landmarks if lm[3] > 0.3]
    ys = [lm[1] for lm in landmarks if lm[3] > 0.3]
    zs = [lm[2] for lm in landmarks if lm[3] > 0.3]
    
    center_x = np.mean(xs) if xs else 0
    center_y = np.mean(ys) if ys else 0
    center_z = np.mean(zs) if zs else 0
    
    for view in views:
        # Create white background
        frame = np.full((h, w, 3), config.bg_color, dtype=np.uint8)
        
        # Convert world coords to screen coords for this view
        # Center the skeleton by subtracting centroid
        screen_points = []
        for lm in landmarks:
            x, y, z, vis = lm
            
            # Center coordinates
            x_centered = x - center_x
            y_centered = y - center_y
            z_centered = z - center_z
            
            sx, sy = world_to_screen(x_centered, y_centered, z_centered, view, w, h, 
                                    scale=config.scale, offset_y=config.offset_y)
            screen_points.append((sx, sy, vis))
        
        # Draw skeleton lines with color by body part
        for start_idx, end_idx in POSE_CONNECTIONS:
            if start_idx >= len(screen_points) or end_idx >= len(screen_points):
                continue
            
            sx1, sy1, vis1 = screen_points[start_idx]
            sx2, sy2, vis2 = screen_points[end_idx]
            
            # Allow points slightly outside frame (don't skip too aggressively)
            margin = 50
            if not (-margin <= sx1 < w+margin and -margin <= sy1 < h+margin and 
                    -margin <= sx2 < w+margin and -margin <= sy2 < h+margin):
                continue
            
            # Get color based on connection type
            avg_vis = (vis1 + vis2) / 2
            if avg_vis < 0.5:
                color = config.low_vis_color
            else:
                conn_type = CONNECTION_TYPES.get((start_idx, end_idx), 'torso')
                if conn_type == 'head':
                    color = config.head_color
                elif conn_type == 'arm':
                    color = config.arm_color
                elif conn_type == 'hand':
                    color = config.hand_color
                else:
                    color = config.torso_color
            
            cv2.line(frame, (sx1, sy1), (sx2, sy2), color, config.line_thickness)
        
        # Draw landmark points
        for i, (sx, sy, vis) in enumerate(screen_points):
            # Allow points slightly outside frame
            margin = 20
            if not (-margin <= sx < w+margin and -margin <= sy < h+margin):
                continue
            
            # Clamp to frame boundaries for drawing
            draw_x = max(0, min(w-1, sx))
            draw_y = max(0, min(h-1, sy))
            
            # Use gray for low visibility points
            if vis < 0.5:
                color = config.low_vis_color
            else:
                color = config.point_color
            
            cv2.circle(frame, (draw_x, draw_y), config.point_radius, color, -1)
        
        # Add view label
        cv2.putText(frame, view.upper(), (10, h - 10), 
                   cv2.FONT_HERSHEY_SIMPLEX, config.font_scale, 
                   config.text_color, config.font_thickness)
        
        view_frames.append(frame)
    
    # Combine views horizontally
    combined = np.hstack(view_frames)
    
    # Add vertical divider lines between views
    h = config.frame_height
    w = config.frame_width
    cv2.line(combined, (w, 0), (w, h), (150, 150, 150), 2)      # After FRONT
    cv2.line(combined, (w*2, 0), (w*2, h), (150, 150, 150), 2)  # After LEFT
    
    return combined

    for frame_idx, frame_features in enumerate(landmarks_array):
        landmarks = extract_landmark_coords(frame_features)
        frame = render_skeleton_frame(landmarks, config)
        
        # Add metadata text to top left
        y_offset = 15
        for line in meta_lines:
            cv2.putText(frame, line, (10, y_offset), 
                       cv2.FONT_HERSHEY_SIMPLEX, config.font_scale,
                       config.text_color, config.font_thickness)
            y_offset += 15
        
        # Add frame counter
        cv2.putText(frame, f"Frame: {frame_idx + 1}/{len(landmarks_array)}", 
                   (10, y_offset + 5), cv2.FONT_HERSHEY_SIMPLEX, 
                   config.font_scale, config.text_color, config.font_thickness)
        
        writer.write(frame)
    
    writer.release()
    return output_path


def render_video_wrapper(args: Tuple[Dict, RenderConfig]) -> Optional[str]:
    """Wrapper for multiprocessing."""
    video_data, config = args
    try:
        return render_video(video_data, config)
    except Exception as e:
        print(f"Error rendering {video_data['metadata']['filename']}: {e}")
        return None

    parser = argparse.ArgumentParser(description='Render skeleton videos from world landmarks')

    parser.add_argument('--output', type=str, default=CONFIG.output_dir,

    parser.add_argument('--limit', type=int, default=None,
                       help='Limit number of videos to render')
    parser.add_argument('--label', type=str, default=None,

    parser.add_argument('--workers', type=int, default=None,

    parser.add_argument('--fps', type=int, default=CONFIG.fps,
                       help='Output video FPS')
    parser.add_argument('--width', type=int, default=RenderConfig.frame_width) 
    parser.add_argument('--height', type=int, default=RenderConfig.frame_height) 
    parser.add_argument('--scale', type=float, default=RenderConfig.scale)
    parser.add_argument('--offset-y', type=float, default=RenderConfig.offset_y) 
     
    
    args = parser.parse_args()

    config = RenderConfig()
    config.npz_path = args.npz
    config.output_dir = args.output
    config.frame_width = args.width
    config.frame_height = args.height
    config.fps = args.fps
    config.scale = args.scale
    config.offset_y = args.offset_y
    
    num_workers = args.workers or max(1, cpu_count() - 2)
    
    print("\n" + "="*70)
    print("SKELETON VIDEO RENDERER")
    print("="*70)
    print(f"Input NPZ: {config.npz_path}")
    print(f"Output dir: {config.output_dir}")
    print(f"Resolution: {config.frame_width * 3} × {config.frame_height}")
    print(f"FPS: {config.fps}")
    print(f"Scale: {config.scale} px/m, Offset Y: {config.offset_y}")
    print(f"Workers: {num_workers}")
    if args.limit:
        print(f"Limit: {args.limit} videos")
    if args.label:
        print(f"Label filter: {args.label}")
    print("="*70 + "\n")
    
    # Load NPZ
    print("Loading world landmarks...")
    if not os.path.exists(config.npz_path):
        print(f"ERROR: NPZ file not found: {config.npz_path}")
        return
    
    data = np.load(config.npz_path, allow_pickle=True)
    
    # Collect videos to render
    labels = ['NoGesture', 'Gesture', 'Move', 'Objman']
    if args.label:
        if args.label not in labels:
            print(f"ERROR: Invalid label '{args.label}'. Must be one of {labels}")
            return
        labels = [args.label]
    
    videos_to_render = []
    
    for label in labels:
        landmarks_key = f'{label}_landmarks'
        metadata_key = f'{label}_metadata'
        
        if landmarks_key not in data:

            continue
        
        landmarks_array = data[landmarks_key]
        metadata_array = data[metadata_key]
        
        print(f"  {label}: {len(landmarks_array)} videos")
        
        for i in range(len(landmarks_array)):
            video_data = {
                'landmarks': landmarks_array[i],
                'metadata': metadata_array[i]
            }
            videos_to_render.append(video_data)
    
    print(f"\nTotal videos to render: {len(videos_to_render)}")
    
    # Apply limit
    if args.limit and args.limit < len(videos_to_render):
        videos_to_render = videos_to_render[:args.limit]
        print(f"Limited to: {len(videos_to_render)} videos")
    
    # Create output directory
    os.makedirs(config.output_dir, exist_ok=True)
    
    # Render videos
    print("\nRendering videos...")
    
    args_list = [(v, config) for v in videos_to_render]
    
    successful = 0
    failed = 0
    
    with Pool(num_workers) as pool:
        results = list(tqdm(
            pool.imap(render_video_wrapper, args_list),
            total=len(args_list),
            desc="Rendering"
        ))
    
    for r in results:
        if r is not None:
            successful += 1
        else:

    print("="*70)

    print(f"Failed: {failed}")
    print(f"Output directory: {config.output_dir}")
    
    # List subdirectories
    for label in labels:

        if os.path.exists(label_dir):
            count = len([f for f in os.listdir(label_dir) if f.endswith('.mp4')])
            print(f"  {label}/: {count} videos")
    
    print("\n")

library(knitr)

py_code_vis <- '
import sys, os, glob
import numpy as np
import pandas as pd

rows = []
for folder, condition in [("Gesture", "gesture"), ("Objman", "objman")]:
    for f in sorted(glob.glob(os.path.join(sys.argv[1], folder, "*_skeleton.npz"))):
        data = np.load(f, allow_pickle=True)
        world = next((data[k] for k in data.files if data[k].ndim == 2 and data[k].shape[1] == 92), None)
        if world is None:
            continue
        # visibility is the 4th value of each landmark (left index = landmark 19, right index = landmark 20)
        rows.append({"dataset": os.path.basename(f).split("_")[0], "condition": condition,
                     "left_index": world[:, 19*4+3].mean(), "right_index": world[:, 20*4+3].mean()})

df = pd.DataFrame(rows)
out = df.groupby(["dataset", "condition"]).agg(
    n_events=("left_index", "size"),
    left_index_mean=("left_index", "mean"), left_index_sd=("left_index", "std"),
    right_index_mean=("right_index", "mean"), right_index_sd=("right_index", "std")).reset_index()
out.to_csv(sys.argv[2], index=False)
'
