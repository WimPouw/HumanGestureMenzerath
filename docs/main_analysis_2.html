# means and SDs of the log-transformed variables, i.e., what scale(log()) does in the models
# we need these to draw the model lines and to get the unscaled exponent
get_scaling <- function(data) {
  list(mx = mean(log(data$n_submovements)), sx = sd(log(data$n_submovements)),
       my = mean(log(data$subm_dur_ms)),    sy = sd(log(data$subm_dur_ms)))
}
sc_ges <- get_scaling(gesdat)
sc_obj <- get_scaling(objman_dat)

# the number of submovements is one value per gesture, so its slope compares gestures with each other
# we split the (log) number of submovements in a part within speaker and a part between speakers:
# x_between = the mean of the speaker, x_within = the deviation of a gesture from the mean of its speaker
# both are divided by the same SD, so x_within + x_between is the scale(log(n_submovements)) of before
add_predictors <- function(data, group = "speaker_id", trans = log, names = c("x_within", "x_between")) {
  x <- trans(data$n_submovements)
  # speaker means are taken over gestures (one value per gesture), not over submovements
  one <- !duplicated(data$event_id)
  speaker_mean <- tapply(x[one], as.character(data[[group]][one]), mean)
  m <- as.numeric(speaker_mean[as.character(data[[group]])])
  data[[names[1]]] <- (x - m) / sd(x)
  data[[names[2]]] <- (m - mean(x)) / sd(x)
  data
}
gesdat <- add_predictors(gesdat)
objman_dat <- add_predictors(objman_dat)

Models

The models that we construct all share that they have as their response variable the log-transformed subm_dur_ms (submovement duration), and as a fixed effect predictor the log-transformed n_submovements (number of submovements, i.e., the number of intervals between speed peaks). Since there are multiple observations of submovement durations per gesture, we include gesture_id as a random effect to account for this non-independence. We follow the modeling approach of (Wascher and Youngblood 2025) as provided here https://masonyoungblood.github.io/crow_efficiency/, and extend it in three ways. The number of submovements is centered within speaker (x_within), with the speaker mean as a separate predictor (x_between), so that the slope of interest compares gestures of the same speaker. The slope is allowed to vary per speaker (by-speaker random slopes). And as there are only six corpora, corpus is a fixed effect.

Note that both variables are scaled after log-transformation, so the reported slopes are standardized. We also report the unscaled slope, which is the exponent of the power law (submovement duration ~ number of submovements^exponent).

NotePer-corpus model fitting
Code 
# one model with a within-speaker slope for each corpus (instead of a separate model per corpus)
# this is the Menzerath model with an interaction between corpus and x_within,
# written such that each corpus directly gets its own intercept and slope
corpus_slope_model <- lmer(scale(log(subm_dur_ms)) ~ 0 + corpus + corpus:x_within + x_between + (1 + x_within|speaker_id) + (1|gesture_id), data = gesdat)

# does the slope differ between corpora? (anova refits both models with ML)
anova_corpus <- anova(gesture_menz_model, corpus_slope_model)
corpus_chisq <- round(anova_corpus$Chisq[2], 2)
corpus_df <- anova_corpus$Df[2]
corpus_p_fmt <- ifelse(anova_corpus$`Pr(>Chisq)`[2] < 0.001, "< 0.001", sprintf("= %.3f", anova_corpus$`Pr(>Chisq)`[2]))

# extract the slope of each corpus
sum_corpus <- coef(summary(corpus_slope_model))
ci_corpus <- confint(corpus_slope_model, method = "Wald")
slope_rows <- grep(":x_within$", rownames(sum_corpus), value = TRUE)

corpus_results_gesture <- data.frame(
  Corpus = sub("^corpus", "", sub(":x_within$", "", slope_rows)),
  beta = sum_corpus[slope_rows, "Estimate"],
  SE = sum_corpus[slope_rows, "Std. Error"],
  CI_low = ci_corpus[slope_rows, 1],
  CI_high = ci_corpus[slope_rows, 2],
  p = sum_corpus[slope_rows, "Pr(>|t|)"]
)
corpus_results_gesture$exponent <- corpus_results_gesture$beta * sc_ges$sy / sc_ges$sx

# number of gestures and speakers per corpus
corpus_n <- gesdat %>%
  group_by(corpus) %>%
  summarise(N_gestures = n_distinct(gesture_id), N_speakers = n_distinct(speaker_id), .groups = "drop")
corpus_results_gesture <- merge(corpus_results_gesture, corpus_n, by.x = "Corpus", by.y = "corpus")

# we test six corpora, so we also give Holm-corrected p-values
corpus_results_gesture$p_holm <- p.adjust(corpus_results_gesture$p, method = "holm")

# format table
corpus_table_gesture <- corpus_results_gesture
row.names(corpus_table_gesture) <- NULL
corpus_table_gesture$beta_fmt <- sprintf("%.3f", corpus_table_gesture$beta)
corpus_table_gesture$SE_fmt <- sprintf("%.3f", corpus_table_gesture$SE)
corpus_table_gesture$CI_fmt <- sprintf("[%.3f, %.3f]", corpus_table_gesture$CI_low, corpus_table_gesture$CI_high)
corpus_table_gesture$p_fmt <- ifelse(corpus_table_gesture$p < 0.001, "< 0.001", sprintf("%.3f", corpus_table_gesture$p))
corpus_table_gesture$p_holm_fmt <- ifelse(corpus_table_gesture$p_holm < 0.001, "< 0.001", sprintf("%.3f", corpus_table_gesture$p_holm))
corpus_table_gesture$exponent_fmt <- sprintf("%.3f", corpus_table_gesture$exponent)

Confound Analysis: Object Manipulation

To assess whether the Menzerath-Altmann relationship is specific to communicative gestures, we repeat the analysis for object manipulation movements which were also annotated in the ECOLANG corpus. We should expect here that these movements follow communicative efficiency principles less (but they might be regulated by general efficiency principles).

# a line with a model slope that goes through the mean of the data points
model_line <- function(x, y, slope, n = 50) {
  x_seq <- seq(min(x), max(x), length.out = n)
  data.frame(x = x_seq, y = mean(y) + slope * (x_seq - mean(x)))
}

Null model

The intervals between speed peaks have to fit within the duration of a gesture. A gesture with more peaks therefore tends to have shorter intervals, even if nothing is abbreviated. To assess this we simulate a null model in which peaks are placed at random within each gesture, keeping the real gesture durations, a minimum gap between peaks that is the same as in the peak detection (160 ms), and a peak rate that is matched per speaker. We then fit the same mixed model on each simulated dataset and compare the observed slope to the slopes that arise under the null. An observed slope that is more negative than the null slopes indicates abbreviation beyond what random peaks would give.

Note that n_sub is the number of intervals between peaks, as in the preprocessing.

library(data.table)

# minimum time between peaks in seconds (PEAK_DIST = 4 frames at 25 fps in the preprocessing)
min_gap <- 0.16

# the same model as in the analysis above:
# x = number of submovements within speaker, xb = speaker mean, by-speaker random slopes,
# and corpus as fixed effect when there is more than one corpus
paper_model <- function(d) {
  d <- copy(d)
  d[, spk := paste(corpus, speaker)]
  d[, lx := log(n_sub)]
  # speaker means are taken over gestures
  d[, m := mean(lx[!duplicated(gid)]), by = spk]
  d[, x := (lx - m) / sd(d$lx)]
  d[, xb := (m - mean(d$lx)) / sd(d$lx)]
  d[, y := scale(log(sub_dur))[, 1]]
  f <- if (uniqueN(d$corpus) > 1) y ~ x + xb + corpus + (1 + x | spk) + (1 | gid) else y ~ x + xb + (1 + x | spk) + (1 | gid)
  lmer(f, data = d, control = lmerControl(calc.derivs = FALSE))
}

sim_gesture <- function(dur, mean_iv, gap = min_gap) {
  for (i in 1:1000) {
    t <- cumsum(gap + rexp(ceiling(dur / gap) + 2, 1 / (mean_iv - gap)))
    t <- t - runif(1) * t[1]
    iv <- diff(t[t <= dur])
    if (length(iv) >= 1) return(iv)
  }
  NULL
}

null_slopes <- function(d, nsim = 1000) {
  gest <- unique(d[, .(corpus, speaker, gid, gesture_dur, n_sub)])
  # n_sub + 1 is the number of peaks
  gest[, mean_iv := sum(gesture_dur) / sum(n_sub + 1), by = .(corpus, speaker)]
  for (r in 1:3) {
    s <- gest[, .(n = length(sim_gesture(gesture_dur, mean_iv))), by = gid]
    gest[s, on = "gid", n_sim := i.n]
    gest[, mean_iv := min_gap + (mean_iv - min_gap) * mean(n_sim) / mean(n_sub), by = .(corpus, speaker)]
  }
  sapply(1:nsim, function(k) {
    s <- gest[, .(sub_dur = sim_gesture(gesture_dur, mean_iv)), by = .(corpus, speaker, gid)]
    s[, n_sub := .N, by = gid]
    unname(fixef(paper_model(s))["x"])
  })
}

# observed
m_g <- paper_model(g)
m_o <- paper_model(o)

# null
null_g <- null_slopes(g)
null_o <- null_slopes(o)

obs_g <- fixef(m_g)[["x"]]
obs_o <- fixef(m_o)[["x"]]
res <- rbind(
  gestures      = c(observed = obs_g, null_mean = mean(null_g),
                    quantile(null_g, c(0.025, 0.975)), p = (1 + sum(null_g <= obs_g)) / 1001),
  manipulations = c(observed = obs_o, null_mean = mean(null_o),
                    quantile(null_o, c(0.025, 0.975)), p = (1 + sum(null_o <= obs_o)) / 1001))

Gesture types

Here we compare the slopes between gesture types. The corpora use different labels, so we first recode them to four general gesture types. For combined labels (e.g., iconic-beat in SAGA) we take the label that comes first. Gestures without a label that fits one of the types are excluded from this analysis. The model has the same structure as the models above, with a within-speaker slope for each gesture type.

NoteGesture type recoding and model fitting

# recoding of the labels of the corpora to general gesture types
# for combined labels (e.g., iconic-beat) we take the label that comes first
recode_type <- function(subtype) {
  # (trimws as one SAGA label has a space at the end: "iconic ")
  first <- sub("-.*$", "", trimws(tolower(as.character(subtype))))
  type <- rep("excluded", length(first))
  type[first %in% c("beat", "beatgest")] <- "beat"
  type[first %in% c("repgest", "iconic", "metaphoric")] <- "representational"
  type[first %in% c("point", "deictic")] <- "pointing"
  type[first %in% c("praggest", "discourse")] <- "pragmatic"
  factor(type, levels = c("beat", "representational", "pointing", "pragmatic", "excluded"))
}
