Null model: Menzerath’s law from random peaks (simulated data)

Here we check the null model that is used at the end of the analysis pages. The simulated data has the same structure as our dataset (corpora, speakers, number of gestures), and follows the preprocessing: peaks are at least 160 ms apart, n_sub is the number of intervals between peaks, and a gesture needs at least one interval.

Simulated data

With b = 0, nothing links submovement duration to the number of submovements. With b < 0, gestures with more submovements are sped up, i.e., a real Menzerath effect.

Code 
make_data <- function(b = 0, gap = min_gap) {
  n_spk  <- c(ECOLANG = 15, MULTISIMO = 18, GESres = 7, SAGA = 4, SAGAplus = 10, ZHUBO = 1)
  n_gest <- c(ECOLANG = 3718, MULTISIMO = 1188, GESres = 1029, SAGA = 687, SAGAplus = 675, ZHUBO = 86)
  g <- rbindlist(lapply(names(n_spk), function(cc) {
    rate <- rlnorm(n_spk[cc], log(2.5), 0.15)              # peaks per second, per speaker
    s <- sample(n_spk[cc], n_gest[cc], replace = TRUE)
    data.table(corpus = cc, speaker = paste0(cc, "_", s), rate = rate[s])
  }))
  g[, `:=`(gid = seq_len(.N), dur = rlnorm(.N, log(1.4), 0.5))]
  g[, {
    t <- cumsum(gap + rexp(ceiling(dur / gap) + 2, 1 / (1 / rate - gap)))
    t <- t - runif(1) * t[1]
    iv <- diff(t[t <= dur])
    if (length(iv) >= 1) {
      f <- (length(iv) / 4)^b
      list(sub_dur = iv * f, gesture_dur = dur * f, n_sub = length(iv))
    }
  }, by = .(corpus, speaker, gid)]
}

# the same model as on the analysis pages:
# x = number of submovements within speaker, xb = speaker mean, by-speaker random slopes, corpus as fixed effect
add_predictors <- function(d) {
  d <- copy(d)
  d[, spk := paste(corpus, speaker)]
  d[, lx := log(n_sub)]
  # speaker means are taken over gestures
  d[, m := mean(lx[!duplicated(gid)]), by = spk]
  d[, x := (lx - m) / sd(d$lx)]
  d[, xb := (m - mean(d$lx)) / sd(d$lx)]
  d[, y := scale(log(sub_dur))[, 1]]
  d
}

paper_model <- function(d) {
  lmer(y ~ x + xb + corpus + (1 + x | spk) + (1 | gid), data = add_predictors(d),
       control = lmerControl(calc.derivs = FALSE))
}

d0 <- make_data(b = 0)
d1 <- make_data(b = -0.1)

Null model

sim_gesture <- function(dur, mean_iv, gap = min_gap) {
  for (i in 1:1000) {
    t <- cumsum(gap + rexp(ceiling(dur / gap) + 2, 1 / (mean_iv - gap)))
    t <- t - runif(1) * t[1]
    iv <- diff(t[t <= dur])
    if (length(iv) >= 1) return(iv)
  }
  NULL
}

null_slopes <- function(d, nsim = 100) {
  gest <- unique(d[, .(corpus, speaker, gid, gesture_dur, n_sub)])
  # n_sub + 1 is the number of peaks
  gest[, mean_iv := sum(gesture_dur) / sum(n_sub + 1), by = speaker]
  for (r in 1:3) {
    s <- gest[, .(n = length(sim_gesture(gesture_dur, mean_iv))), by = gid]
    gest[s, on = "gid", n_sim := i.n]
    gest[, mean_iv := min_gap + (mean_iv - min_gap) * mean(n_sim) / mean(n_sub), by = speaker]
  }
  sapply(1:nsim, function(k) {
    s <- gest[, .(sub_dur = sim_gesture(gesture_dur, mean_iv)), by = .(corpus, speaker, gid)]
    s[, n_sub := .N, by = gid]
    unname(fixef(paper_model(s))["x"])
  })
}

per_corpus <- function(d) {
  d <- add_predictors(d)
  # one model with a within-speaker slope for each corpus
  mod <- lmer(y ~ 0 + corpus + corpus:x + xb + (1 + x | spk) + (1 | gid), data = d)
  cf <- coef(summary(mod))
  rows <- grep(":x$", rownames(cf), value = TRUE)
  out <- data.table(corpus = sub("^corpus", "", sub(":x$", "", rows)),
                    beta = cf[rows, "Estimate"], p = cf[rows, "Pr(>|t|)"])
  out[d[, .(gestures = uniqueN(gid)), by = corpus], on = "corpus", gestures := i.gestures]
  out[, singular := isSingular(mod)]
  out
}
